    - `probability: 0.5` with seed 42 over 40 calls gives the same faulted call-number list on two independent wrappers; seed 7 gives a different list; probability 0 never fires; 1 always fires
    - A rule naming a method the port does not have throws synchronously at wrap time (message contains the method name)
    - FaultRule.parse rejects `{ method:'x', fault:{ type:'http' } }` and fills default retryAfterMs 50 and status 500

```ts
export const FaultSpec = z.discriminatedUnion('type', [
  z.object({ type: z.literal('rate_limit'), retryAfterMs: z.number().int().nonnegative().default(50) }),
  z.object({ type: z.literal('server_error'), status: z.union([z.literal(500), z.literal(502), z.literal(503)]).default(500) }),
  z.object({ type: z.literal('ghost_write'), status: z.union([z.literal(500), z.literal(502), z.literal(503)]).default(500) }), // commit, THEN fail
  z.object({ type: z.literal('lying_success') }),                                   // report ok, persist nothing
  z.object({ type: z.literal('latency'), ms: z.number().int().nonnegative().max(2000) })
]);
export const FaultRule = z.object({
  method: z.string().min(1),                                   // exact port method name, or '*'
  fault: FaultSpec,
  calls: z.array(z.number().int().positive()).optional(),      // 1-based per-method call numbers
  probability: z.number().min(0).max(1).optional(),            // seeded PRNG
  maxTimes: z.number().int().positive().optional()
});
export type FaultSpec = z.infer<typeof FaultSpec>; export type FaultRule = z.infer<typeof FaultRule>; export type FaultRuleInput = z.input<typeof FaultRule>;
export type FaultType = FaultSpec['type'];
export interface FaultEvent { app?: string; method: string; call: number; type: FaultType }
export interface StateRollback<S = unknown> { snapshot(): S; restore(snapshot: S): void } // World satisfies this structurally
export interface FaultOptions { seed?: number; app?: string; rollback?: StateRollback<any>; sleep?: (ms: number) => Promise<void>; onFault?: (e: FaultEvent) => void }
export function withFaults<T extends object>(port: T, rules: FaultRuleInput[], opts?: FaultOptions): T;
export function withFaultsAll<C extends Connectors>(connectors: C, rules: Partial<Record<keyof Connectors, FaultRuleInput[]>>, opts?: Omit<FaultOptions, 'app'>): C;
```
Implementation rules:
- `const parsed = FaultRule.array().parse(rules)`. For each rule where `method !== '*'` and `typeof (port as any)[method] !== 'function'`, `throw new Error(`withFaults: port has no method "${method}"`)`.
- Local PRNG inside the call: mulberry32 over `opts.seed ?? 1` (write it as a small local function, no dependency). `counts: Record<string, number>` and `fired: number[]` are also local.
- Build `out = {}`. For each own key of `port`: non-functions are copied as-is. For functions: `out[key] = async (...args) => { const call = (counts[key] = (counts[key] ?? 0) + 1); let terminal: FaultSpec | undefined; for (const [i, r] of parsed.entries()) { if (r.method !== '*' && r.method !== key) continue; if (r.calls && !r.calls.includes(call)) continue; if (r.maxTimes !== undefined && (fired[i] ?? 0) >= r.maxTimes) continue; if (r.probability !== undefined && !(rng() < r.probability)) continue; fired[i] = (fired[i] ?? 0) + 1; opts.onFault?.({ app: opts.app, method: key, call, type: r.fault.type }); if (r.fault.type === 'latency') { await sleep(r.fault.ms); continue; } terminal = r.fault; break; } ... }`. Evaluate `probability` LAST so the PRNG is consumed only by otherwise-matching calls (keeps sequences stable).
- Terminal handling: rate_limit → `throw new ConnectorError('rate_limit', `${opts.app ?? 'port'}.${key}: 429 Too Many Requests`, { status: 429, retryAfterMs })`; server_error → `throw new ConnectorError('server', `${...}.${key}: ${status} Internal Server Error`, { status })`; ghost_write → `await fn(...args)` (propagate its error if it throws), then throw the same server error with message suffix `(write was committed)`; lying_success → no rollback → `throw new Error('withFaults: lying_success requires opts.rollback (pass the World)')`; else `const snap = rollback.snapshot(); const res = await fn(...args); rollback.restore(snap); return res;`. With no terminal fault → `return fn(...args)`.
- Default `sleep = (ms) => new Promise(r => setTimeout(r, ms))`.
- Call the original with `fn.apply(port, args)` so `this`-free plain objects and any class-based real ports both work.
- `withFaultsAll`: `const out = { ...connectors }`. For each app key with a non-empty rule list, `out[app] = withFaults(connectors[app], rules[app], { ...opts, app, seed: ((opts?.seed ?? 1) * 31 + APP_SALT[app]) >>> 0 })` where `APP_SALT = { gmail: 1, calendar: 2, docs: 3, notion: 4, github: 5, hf: 6 }` (a readonly const). Return `out as C`. This preserves extra bundle props like `mode` and `world`.
