```ts
export interface RetryInfo { attempt: number; error: unknown; delayMs: number }
export interface RetryOptions {
  maxAttempts: number; baseDelayMs: number; maxDelayMs: number;
  sleep: (ms: number) => Promise<void>;
  isRetryable?: (err: unknown) => boolean;               // default isRetryableError
  onRetry?: (info: RetryInfo) => void;                   // called before sleeping
  beforeRetry?: (info: RetryInfo) => Promise<'retry' | 'stop'>; // called after sleeping; 'stop' ends the loop (e.g. findByKey found the ghost write)
}
export type RetryResult<T> = { ok: true; value: T; attempts: number } | { ok: false; error: unknown; attempts: number; stopped: boolean };
export function isRetryableError(err: unknown): boolean;   // err instanceof ConnectorError && ['rate_limit','server','timeout'].includes(err.kind)
export function backoffDelay(attempt: number, err: unknown, o: { baseDelayMs: number; maxDelayMs: number }): number; // min(maxDelayMs, retryAfterMs ?? baseDelayMs * 2 ** (attempt - 1)); deterministic, no jitter
export function toErrorInfo(err: unknown): ConnectorErrorInfo;  // ConnectorError → { kind, message, status?, retryAfterMs? } (omit undefined); other → { kind: 'server', message }
export async function withRetry<T>(fn: (attempt: number) => Promise<T>, o: RetryOptions): Promise<RetryResult<T>>;
```
Loop `attempt = 1..maxAttempts`:
- try `return { ok: true, value: await fn(attempt), attempts: attempt }`.
- On catch, if `!isRetryable(err) || attempt === maxAttempts`, return `{ ok: false, error: err, attempts: attempt, stopped: false }`.
- Otherwise compute the delay, call `onRetry`, then `await sleep(delay)`.
- `if (o.beforeRetry && (await o.beforeRetry(info)) === 'stop') return { ok: false, error: err, attempts: attempt, stopped: true }`.

Header comment: a generic retry around creates is only safe because the executor's beforeRetry re-checks findByKey (Pitfall 6).
